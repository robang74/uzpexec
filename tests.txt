bin=uzpexec
LS=$(command -v ls)
DD() { dd status=none "$@"; }
retprt() { printf "\tret:$?\n\n"; }
HI() { printf '#!/bin/sh\necho Hello World!\n' | gzip -c; }
do_script() { sed -e 's,\x00\(/bin/sh\),\x01\1,' -i $bin; }
no_script() { sed -e 's,\x01\(/bin/sh\),\x00\1,' -i $bin; }

if test "x${1:-}" = "x"; then
  make clean $bin || { printf "nasm failed\n\n"; exit; }
else
  echo
fi

echo "Code size with EOF string:"
n=$(grep --color=never -abo "elf_eof" $bin | cut -f1 -d:)
printf "\t%d bytes\n\n" $(( ${n:--8} + 8 ))

echo "Strings output:"
( exec 2>&1
  echo "$(strings $bin)" | sed -e "s/^/\t/" | tee /proc/self/fd/2 |
  grep -qe "([cC]) .*robang74 v[0-9.]\{4\}" ||{
    printf "\nWARNING:\n"
    printf "\tAuthorship isn't allowed to be changed or removed"
    printf "\n\n"
    exit 1
  }
) || { make clean >&- 2>&1; exit 1; }
( exec 2>&1
  echo
  echo Binary size check:
  printf "\t%s\n" "$(du -b $bin)"

  echo
  echo "== TESTS TO PASS (x5) =="
  echo

  gzip -c $LS | $bin -1 ./$bin
  retprt

  do_script #####################

  HI | $bin
  retprt

  { cat $bin; HI; } > $bin.uzp &&
             chmod +x $bin.uzp
  ( exec $bin.uzp ) ; retprt

  no_script #####################

  { cat $bin; gzip -c $LS; } | DD of=ls.gz.elf &&
  chmod +x ls.gz.elf && ls.gz.elf -1 ls.gz.elf
  retprt

  cat $LS | $bin -1 $bin
  retprt

  echo "== TESTS TO FAIL (x3) =="

  cat $bin $LS | DD of=ls.elf &&
  chmod +x ls.elf && ls.elf -1 ls.elf
  retprt

  $bin <&-
  retprt

  echo try to run a compressed shell script
  HI | $bin
  retprt

) | tee     tests.res
printf "Tests final result: "
sha1sum     tests.res | cut -d' ' -f1 |
sed "s/f6806c33616cf04d08f02136b7fe8ab35e111f49/$bin OK/" |
tee /proc/self/fd/2 | grep -qe " OK$" || printf "\t%s FAILED\n" $bin

################################################################################
echo;exit
################################################################################

bin=uskex
nasm -O2 -s -f bin $bin.asm -o $bin &&
chmod +x $bin && du -b $bin
# 512     uskex
cat $(which ls) >> $bin &&
./$bin -al $bin; echo $?
# -rwxrwxr-x 1 roberto roberto 138728 Jun 23 14:19 uskex
# 0

bin=uskat
nasm -O2 -s -f bin $bin.asm -o $bin &&
chmod +x $bin && du -b $bin
# 512     uskat
echo 'Hello wonderful World!' >> $bin &&
./$bin -al $bin; echo $?
# Hello wonderful World!
# 0
